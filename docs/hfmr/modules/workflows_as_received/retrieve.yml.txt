name: Public permit acquisition
on:
  workflow_dispatch:
  schedule:
    - cron: '23 11 * * 1'
permissions:
  contents: read
jobs:
  acquire:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Unit tests
        run: python -m unittest discover -s tests -v
      - name: Acquire public Columbus permit records
        run: python -m hfmr.auto_retrieve acquire --output evidence --number 1546 --street HIGH --start 2000 --end 2010
      - name: Verify artifacts
        if: always()
        run: python -m hfmr.auto_retrieve verify --output evidence
      - name: Upload retrieval evidence
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: case001-permit-evidence-${{ github.run_id }}
          path: evidence/
          retention-days: 30
          if-no-files-found: warn
